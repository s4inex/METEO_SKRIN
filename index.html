// ========================================================== // --- GPU NOISE FLOW FIELD SIMULATION ENGINE (WEBGL2) ----- // ========================================================== const boidsCanvas = document.getElementById('boidsCanvas'); const gl = boidsCanvas.getContext('webgl2', { preserveDrawingBuffer: true, alpha: false }); if (!gl || !gl.getExtension('EXT_color_buffer_float')) { console.error("WebGL2 Float Textures unsupported."); } // Scaled down to 64x64 grid = 4,096 Flow Agents const TEX_WIDTH = 64; const TEX_HEIGHT = 64; const NUM_BOIDS = TEX_WIDTH * TEX_HEIGHT; const updateVS = `#version 300 es out vec2 v_texCoord; void main() { vec2 pos[4] = vec2[](vec2(-1.0,-1.0), vec2(1.0,-1.0), vec2(-1.0,1.0), vec2(1.0,1.0)); vec2 tex[4] = vec2[](vec2(0.0,0.0), vec2(1.0,0.0), vec2(0.0,1.0), vec2(1.0,1.0)); gl_Position = vec4(pos[gl_VertexID], 0.0, 1.0); v_texCoord = tex[gl_VertexID]; }`; const velUpdateFS = `#version 300 es precision highp float; uniform sampler2D u_posTex; uniform sampler2D u_velTex; uniform float u_time; in vec2 v_texCoord; out vec4 fragColor; float hash(vec3 p) { p = fract(p * vec3(443.8975, 397.2973, 491.1871)); p += dot(p.xyz, p.yzx + 19.19); return fract(p.x * p.y * p.z); } float noise(vec3 p) { vec3 i = floor(p); vec3 f = fract(p); f = f*f*(3.0-2.0*f); return mix(mix(mix(hash(i+vec3(0,0,0)), hash(i+vec3(1,0,0)), f.x), mix(hash(i+vec3(0,1,0)), hash(i+vec3(1,1,0)), f.x), f.y), mix(mix(hash(i+vec3(0,0,1)), hash(i+vec3(1,0,1)), f.x), mix(hash(i+vec3(0,1,1)), hash(i+vec3(1,1,1)), f.x), f.y), f.z); } void main() { vec2 pos = texture(u_posTex, v_texCoord).xy; vec2 vel = texture(u_velTex, v_texCoord).xy; // Higher multiplier (0.008) makes the noise field patterns smaller and tighter vec3 noiseCoord = vec3(pos * 0.008, u_time * 0.15); float n = noise(noiseCoord); float angle = n * 3.14159265359 * 4.0; float maxSpeed = 2.0; vec2 targetVel = vec2(cos(angle), sin(angle)) * maxSpeed; vel = mix(vel, targetVel, 0.06); fragColor = vec4(vel, 0.0, 1.0); }`; const posUpdateFS = `#version 300 es precision highp float; uniform sampler2D u_posTex; uniform sampler2D u_velTex; uniform vec2 u_screenSize; in vec2 v_texCoord; out vec4 fragColor; void main() { vec2 pos = texture(u_posTex, v_texCoord).xy; vec2 vel = texture(u_velTex, v_texCoord).xy; pos += vel; if(pos.x < -10.0) pos.x = u_screenSize.x + 10.0; if(pos.x > u_screenSize.x + 10.0) pos.x = -10.0; if(pos.y < -10.0) pos.y = u_screenSize.y + 10.0; if(pos.y > u_screenSize.y + 10.0) pos.y = -10.0; fragColor = vec4(pos, 0.0, 1.0); }`; const trailVS = `#version 300 es void main() { vec2 pos[6] = vec2[](vec2(-1.0,-1.0), vec2(1.0,-1.0), vec2(-1.0,1.0), vec2(-1.0,1.0), vec2(1.0,-1.0), vec2(1.0,1.0)); gl_Position = vec4(pos[gl_VertexID], 0.0, 1.0); }`; const trailFS = `#version 300 es precision highp float; uniform vec4 u_bgColor; out vec4 fragColor; void main() { fragColor = u_bgColor; }`; const renderVS = `#version 300 es uniform sampler2D u_posTex; uniform sampler2D u_velTex; uniform vec2 u_texSize; uniform vec2 u_screenSize; void main() { int agentID = gl_VertexID / 2; int vertexID = gl_VertexID % 2; // Updated to read from the 64-wide layout structure float x = float(agentID % 64); float y = float(agentID / 64); vec2 texCoord = (vec2(x, y) + 0.5) / u_texSize; vec2 pos = texture(u_posTex, texCoord).xy; vec2 vel = texture(u_velTex, texCoord).xy; vec2 finalPos = pos; if (vertexID == 1) { finalPos -= vel * 2.0; } vec2 clipSpace = (finalPos / u_screenSize) * 2.0 - 1.0; gl_Position = vec4(clipSpace.x, -clipSpace.y, 0.0, 1.0); }`; const renderFS = `#version 300 es precision highp float; out vec4 fragColor; void main() { fragColor = vec4(0.38, 0.68, 1.0, 0.35); }`; function createShader(gl, type, source) { const shader = gl.createShader(type); gl.shaderSource(shader, source); gl.compileShader(shader); if (!gl.getShaderParameter(shader, gl.COMPILE_STATUS)) { gl.deleteShader(shader); return null; } return shader; } function createProgram(gl, vsSrc, fsSrc) { const vs = createShader(gl, gl.VERTEX_SHADER, vsSrc); const fs = createShader(gl, gl.FRAGMENT_SHADER, fsSrc); const program = gl.createProgram(); gl.attachShader(program, vs); gl.attachShader(program, fs); gl.linkProgram(program); return program; } const velUpdateProg = createProgram(gl, updateVS, velUpdateFS); const posUpdateProg = createProgram(gl, updateVS, posUpdateFS); const trailProg = createProgram(gl, trailVS, trailFS); const renderProg = createProgram(gl, renderVS, renderFS); function createGPGPUBuffer(gl, width, height, initialData) { const tex = gl.createTexture(); gl.bindTexture(gl.TEXTURE_2D, tex); gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA32F, width, height, 0, gl.RGBA, gl.FLOAT, initialData); gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.NEAREST); gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.NEAREST); gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE); gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE); const fb = gl.createFramebuffer(); gl.bindFramebuffer(gl.FRAMEBUFFER, fb); gl.framebufferTexture2D(gl.FRAMEBUFFER, gl.COLOR_ATTACHMENT0, gl.TEXTURE_2D, tex, 0); return { tex, fb }; } const posData = new Float32Array(NUM_BOIDS * 4); const velData = new Float32Array(NUM_BOIDS * 4); for (let i = 0; i < NUM_BOIDS; i++) { posData[i * 4 + 0] = Math.random() * window.innerWidth; posData[i * 4 + 1] = Math.random() * window.innerHeight; const angle = Math.random() * Math.PI * 2; const speed = Math.random() * 1.2 + 0.8; velData[i * 4 + 0] = Math.cos(angle) * speed; velData[i * 4 + 1] = Math.sin(angle) * speed; } let bufPosA = createGPGPUBuffer(gl, TEX_WIDTH, TEX_HEIGHT, posData); let bufPosB = createGPGPUBuffer(gl, TEX_WIDTH, TEX_HEIGHT, null); let bufVelA = createGPGPUBuffer(gl, TEX_WIDTH, TEX_HEIGHT, velData); let bufVelB = createGPGPUBuffer(gl, TEX_WIDTH, TEX_HEIGHT, null); const velLocs = { posTex: gl.getUniformLocation(velUpdateProg, "u_posTex"), velTex: gl.getUniformLocation(velUpdateProg, "u_velTex"), time: gl.getUniformLocation(velUpdateProg, "u_time") }; const posLocs = { posTex: gl.getUniformLocation(posUpdateProg, "u_posTex"), velTex: gl.getUniformLocation(posUpdateProg, "u_velTex"), screenSize: gl.getUniformLocation(posUpdateProg, "u_screenSize") }; const renderLocs = { posTex: gl.getUniformLocation(renderProg, "u_posTex"), velTex: gl.getUniformLocation(renderProg, "u_velTex"), texSize: gl.getUniformLocation(renderProg, "u_texSize"), screenSize: gl.getUniformLocation(renderProg, "u_screenSize") }; const trailBgColorLoc = gl.getUniformLocation(trailProg, "u_bgColor"); function resizeCanvas() { boidsCanvas.width = window.innerWidth; boidsCanvas.height = window.innerHeight; gl.viewport(0, 0, boidsCanvas.width, boidsCanvas.height); gl.clearColor(31 / 255, 41 / 255, 55 / 255, 1.0); gl.clear(gl.COLOR_BUFFER_BIT); } window.addEventListener('resize', resizeCanvas); resizeCanvas(); const startTime = performance.now(); function animateBoids() { const elapsedSeconds = (performance.now() - startTime) * 0.001; gl.bindFramebuffer(gl.FRAMEBUFFER, bufVelB.fb); gl.viewport(0, 0, TEX_WIDTH, TEX_HEIGHT); gl.useProgram(velUpdateProg); gl.activeTexture(gl.TEXTURE0); gl.bindTexture(gl.TEXTURE_2D, bufPosA.tex); gl.activeTexture(gl.TEXTURE1); gl.bindTexture(gl.TEXTURE_2D, bufVelA.tex); gl.uniform1i(velLocs.posTex, 0); gl.uniform1i(velLocs.velTex, 1); gl.uniform1f(velLocs.time, elapsedSeconds); gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4); gl.bindFramebuffer(gl.FRAMEBUFFER, bufPosB.fb); gl.viewport(0, 0, TEX_WIDTH, TEX_HEIGHT); gl.useProgram(posUpdateProg); gl.activeTexture(gl.TEXTURE0); gl.bindTexture(gl.TEXTURE_2D, bufPosA.tex); gl.activeTexture(gl.TEXTURE1); gl.bindTexture(gl.TEXTURE_2D, bufVelB.tex); gl.uniform1i(posLocs.posTex, 0); gl.uniform1i(posLocs.velTex, 1); gl.uniform2f(posLocs.screenSize, boidsCanvas.width, boidsCanvas.height); gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4); gl.bindFramebuffer(gl.FRAMEBUFFER, null); gl.viewport(0, 0, boidsCanvas.width, boidsCanvas.height); gl.useProgram(trailProg); gl.uniform4f(trailBgColorLoc, 9 / 255, 19 / 255, 33 / 255, 0.045); // Marginally higher opacity for tighter trailing lines gl.enable(gl.BLEND); gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA); gl.drawArrays(gl.TRIANGLES, 0, 6); gl.useProgram(renderProg); gl.activeTexture(gl.TEXTURE0); gl.bindTexture(gl.TEXTURE_2D, bufPosB.tex); gl.activeTexture(gl.TEXTURE1); gl.bindTexture(gl.TEXTURE_2D, bufVelB.tex); gl.uniform1i(renderLocs.posTex, 0); gl.uniform1i(renderLocs.velTex, 1); gl.uniform2f(renderLocs.texSize, TEX_WIDTH, TEX_HEIGHT); gl.uniform2f(renderLocs.screenSize, boidsCanvas.width, boidsCanvas.height); gl.drawArrays(gl.LINES, 0, NUM_BOIDS * 2); let tempP = bufPosA; bufPosA = bufPosB; bufPosB = tempP; let tempV = bufVelA; bufVelA = bufVelB; bufVelB = tempV; requestAnimationFrame(animateBoids); } animateBoids();
